> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App Flutter para Android (Galaxy Z Fold 8) con flashcards, practica de trazo y vocabulario de hiragana, extensible a otros modulos de estudio.

**Architecture:** UI (Riverpod) -> motores/sesiones -> `StudyRepository` -> SQLite (Drift) sembrada desde codigo Dart. Una sola base de codigo con dos disposiciones (slab < 600 dp, extendida >= 600 dp) elegidas por ancho de ventana.

**Tech Stack:** Flutter (solo Android), flutter_riverpod ^2.6.1, go_router, drift + drift_flutter + drift_dev + build_runner, flutter_tts, fuentes locales (Zen Maru Gothic, Shippori Mincho).

**Spec:** `docs/superpowers/specs/2026-10-12-petrufc-design.md`. Mockups: https://claude.ai/artifact/51QinWbKsXVbQiku2H3w36

## Global Constraints

- Solo Android; nombre del proyecto `petru_fc`, org `com.petru`, etiqueta de la app "Petru FC".
- 100% offline: las fuentes van empaquetadas en `assets/fonts/` (nada de `google_fonts` en runtime).
- Disposicion extendida cuando `MediaQuery.sizeOf(context).width >= 600`.
- Paleta PetruTheme Dark exacta: Base `#13171B`, Mantle `#0C0E11`, Crust `#060809`, Surface0 `#1F262D`, Surface1 `#2C363F`, Surface2 `#3B4754`, Text `#EEF0F2`, Subtext0 `#A4ADB7`, Flamingo `#FC8783`, Pink `#FC83DC`, Teal `#83FCE8`, Green `#8DFC83`, Red `#FC83A5`; lienzo `#F7F7F8`, trazo `#1D1D20`, guia `#D1D0DD`.
- Texto de UI en espanol latinoamericano neutro. Vocabulario solo en hiragana (sin kanji ni katakana).
- `items.reading` es unico por item kana: ぢ=`di`, づ=`du`, を=`wo`.
- La columna del texto japones se llama `japanese` (no `text`: choca con el builder `text()` de Drift).
- Sin pruebas automatizadas (decision del usuario). Verificacion: `flutter analyze` ("No issues found!") y prueba manual.
- El directorio no es un repo git: no hay pasos de commit. Si el usuario corre `git init`, commitear al final de cada tarea.

import 'package:drift/drift.dart';

import 'database.dart';

class StudyRepository {
  StudyRepository(this.db);
  final AppDatabase db;

  Future<List<StudyModule>> modules() => (db.select(db.modules)
        ..orderBy([(t) => OrderingTerm.asc(t.sortOrder)]))
      .get();

  Future<List<ItemGroup>> groups(String moduleId, String kind) =>
      (db.select(db.itemGroups)
            ..where((g) => g.moduleId.equals(moduleId) & g.kind.equals(kind))
            ..orderBy([(g) => OrderingTerm.asc(g.sortOrder)]))
          .get();

  Future<List<Item>> items(String moduleId, Iterable<String> groupIds) =>
      (db.select(db.items)
            ..where((i) => i.moduleId.equals(moduleId) & i.groupId.isIn(groupIds)))
          .get();

  Future<Set<String>> activeGroups(String moduleId) async {
    final row = await (db.select(db.settings)
          ..where((s) => s.key.equals('active_groups:$moduleId')))
        .getSingleOrNull();
    final v = row?.value ?? '';
    return v.isEmpty ? <String>{} : v.split(',').toSet();
  }

  Future<void> saveActiveGroups(String moduleId, Set<String> ids) =>
      db.into(db.settings).insertOnConflictUpdate(SettingsCompanion.insert(
          key: 'active_groups:$moduleId', value: ids.join(',')));

  Future<void> recordAnswer(int itemId, bool correct) => db.transaction(() async {
        final cur = await (db.select(db.reviewStats)
              ..where((r) => r.itemId.equals(itemId)))
            .getSingleOrNull();
        await db.into(db.reviewStats).insertOnConflictUpdate(
              ReviewStatsCompanion.insert(
                itemId: itemId,
                attempts: Value((cur?.attempts ?? 0) + 1),
                correct: Value((cur?.correct ?? 0) + (correct ? 1 : 0)),
                lastSeen: DateTime.now(),
              ),
            );
      });
}

class TtsService {
  final _tts = FlutterTts();
  bool _ready = false;

  /// Devuelve un mensaje para mostrar al usuario si falla, o null si habla.
  Future<String?> speak(String text) async {
    try {
      if (!_ready) {
        final available = await _tts.isLanguageAvailable('ja-JP');
        if (available != true) {
          return 'Falta la voz japonesa. Instálala en Ajustes > Salida de texto a voz.';
        }
        await _tts.setLanguage('ja-JP');
        await _tts.setSpeechRate(0.4);
        _ready = true;
      }
      await _tts.stop();
      await _tts.speak(text);
      return null;
    } catch (_) {
      return 'No se pudo reproducir el audio.';
    }
  }
}

import 'dart:math';

import '../../core/data/database.dart';

enum QuizMode {
  kanaToRomaji('Hiragana → Romaji', 'あ→a', 'kana', true),
  romajiToKana('Romaji → Hiragana', 'a→あ', 'kana', false),
  wordToMeaning('Hiragana → Español', 'はな→Nariz', 'word', true),
  meaningToWord('Español → Hiragana', 'Nariz→はな', 'word', false);

  const QuizMode(this.label, this.badge, this.kind, this.promptIsJapanese);
  final String label, badge, kind;
  final bool promptIsJapanese;

  bool get answersAreJapanese => !promptIsJapanese;

  String prompt(Item i) => switch (this) {
        kanaToRomaji || wordToMeaning => i.japanese,
        romajiToKana => i.reading,
        meaningToWord => i.meaningEs!,
      };

  String answer(Item i) => switch (this) {
        kanaToRomaji => i.reading,
        romajiToKana || meaningToWord => i.japanese,
        wordToMeaning => i.meaningEs!,
      };
}

class QuizQuestion {
  const QuizQuestion(this.item, this.prompt, this.options, this.correct);
  final Item item;
  final String prompt;
  final List<String> options;
  final String correct;
}

class QuizEngine {
  QuizEngine(this.pool, this.mode, [Random? rng]) : _rng = rng ?? Random();
  final List<Item> pool;
  final QuizMode mode;
  final Random _rng;
  int? _lastId;

  QuizQuestion next() {
    Item target;
    do {
      target = pool[_rng.nextInt(pool.length)];
    } while (pool.length > 1 && target.id == _lastId);
    _lastId = target.id;
    final correct = mode.answer(target);
    final others = pool.map(mode.answer).where((a) => a != correct).toSet().toList()
      ..shuffle(_rng);
    final options = [correct, ...others.take(3)]..shuffle(_rng);
    return QuizQuestion(target, mode.prompt(target), options, correct);
  }
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../core/data/providers.dart';
import 'quiz_engine.dart';

typedef QuizArgs = (String moduleId, QuizMode mode);

class QuizState {
  const QuizState({
    required this.question,
    this.selected,
    this.correct = 0,
    this.wrong = 0,
  });
  final QuizQuestion question;
  final String? selected;
  final int correct, wrong;

  bool get answered => selected != null;
}

class QuizSession extends AutoDisposeFamilyAsyncNotifier<QuizState, QuizArgs> {
  late QuizEngine _engine;

  @override
  Future<QuizState> build(QuizArgs arg) async {
    final (moduleId, mode) = arg;
    final repo = ref.read(repoProvider);
    final Set<String> ids;
    if (mode.kind == 'kana') {
      ids = await ref.read(activeGroupsProvider(moduleId).future);
    } else {
      ids = (await repo.groups(moduleId, 'word')).map((g) => g.id).toSet();
    }
    _engine = QuizEngine(await repo.items(moduleId, ids), mode);
    return QuizState(question: _engine.next());
  }

  void answer(String option) {
    final s = state.requireValue;
    if (s.answered) return;
    final ok = option == s.question.correct;
    ref.read(repoProvider).recordAnswer(s.question.item.id, ok);
    state = AsyncData(QuizState(
      question: s.question,
      selected: option,
      correct: s.correct + (ok ? 1 : 0),
      wrong: s.wrong + (ok ? 0 : 1),
    ));
  }

  void next() {
    final s = state.requireValue;
    state = AsyncData(QuizState(
      question: _engine.next(),
      correct: s.correct,
      wrong: s.wrong,
    ));
  }
}

final quizSessionProvider = AsyncNotifierProvider.autoDispose
    .family<QuizSession, QuizState, QuizArgs>(QuizSession.new);

class PracticeState {
  const PracticeState({
    required this.item,
    this.strokes = const [],
    this.checked = false,
  });
  final Item item;
  final List<List<Offset>> strokes; // coordenadas normalizadas 0..1
  final bool checked;

  PracticeState copyWith({List<List<Offset>>? strokes, bool? checked}) =>
      PracticeState(
        item: item,
        strokes: strokes ?? this.strokes,
        checked: checked ?? this.checked,
      );
}

class PracticeSession
    extends AutoDisposeFamilyAsyncNotifier<PracticeState, String> {
  late List<Item> _pool;
  int? _lastId;
  final _rng = Random();

  Item _pick() {
    Item i;
    do {
      i = _pool[_rng.nextInt(_pool.length)];
    } while (_pool.length > 1 && i.id == _lastId);
    _lastId = i.id;
    return i;
  }

  @override
  Future<PracticeState> build(String arg) async {
    final ids = await ref.read(activeGroupsProvider(arg).future);
    _pool = await ref.read(repoProvider).items(arg, ids);
    return PracticeState(item: _pick());
  }

  void startStroke(Offset p) {
    final s = state.valueOrNull;
    if (s == null || s.checked) return;
    state = AsyncData(s.copyWith(strokes: [
      ...s.strokes,
      [p]
    ]));
  }

  void addPoint(Offset p) {
    final s = state.valueOrNull;
    if (s == null || s.checked || s.strokes.isEmpty) return;
    final last = [...s.strokes.last, p];
    state = AsyncData(
        s.copyWith(strokes: [...s.strokes.sublist(0, s.strokes.length - 1), last]));
  }

  void clear() {
    final s = state.valueOrNull;
    if (s != null) state = AsyncData(s.copyWith(strokes: const []));
  }

  void check() {
    final s = state.valueOrNull;
    if (s != null) state = AsyncData(s.copyWith(checked: true));
  }

  void grade(bool ok) {
    final s = state.requireValue;
    ref.read(repoProvider).recordAnswer(s.item.id, ok);
    state = AsyncData(PracticeState(item: _pick()));
  }
}

final practiceSessionProvider = AsyncNotifierProvider.autoDispose

import 'package:flutter/material.dart';

import '../../app/theme.dart';

class StrokeBoard extends StatelessWidget {
  const StrokeBoard({super.key, required this.strokes, this.onStart, this.onMove});
  final List<List<Offset>> strokes;
  final ValueChanged<Offset>? onStart, onMove;

  @override
  Widget build(BuildContext context) => ClipRRect(
        borderRadius: BorderRadius.circular(32),
        child: LayoutBuilder(builder: (context, c) {
          Offset norm(Offset p) => Offset(
                (p.dx / c.maxWidth).clamp(0.0, 1.0).toDouble(),
                (p.dy / c.maxHeight).clamp(0.0, 1.0).toDouble(),
              );
          return Listener(
            behavior: HitTestBehavior.opaque,
            onPointerDown: onStart == null ? null : (e) => onStart!(norm(e.localPosition)),
            onPointerMove: onMove == null ? null : (e) => onMove!(norm(e.localPosition)),
            child: CustomPaint(painter: _StrokePainter(strokes), size: Size.infinite),
          );
        }),
      );
}

class _StrokePainter extends CustomPainter {
  _StrokePainter(this.strokes);
  final List<List<Offset>> strokes;

  @override
  void paint(Canvas canvas, Size size) {
    canvas.drawRect(Offset.zero & size, Paint()..color = Pt.paper);
    final guide = Paint()
      ..color = Pt.paperLine
      ..strokeWidth = 1.5;
    canvas.drawLine(Offset(size.width / 2, size.height * 0.06),
        Offset(size.width / 2, size.height * 0.94), guide);
    canvas.drawLine(Offset(size.width * 0.07, size.height / 2),
        Offset(size.width * 0.93, size.height / 2), guide);

    final width = (size.shortestSide * 0.03).clamp(3.0, 14.0).toDouble();
    final ink = Paint()
      ..color = Pt.ink
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round
      ..strokeWidth = width;
    Offset at(Offset p) => Offset(p.dx * size.width, p.dy * size.height);
    for (final s in strokes) {
      if (s.length == 1) {
        canvas.drawCircle(at(s.first), width / 2, Paint()..color = Pt.ink);
        continue;
      }
      final path = Path()..moveTo(at(s.first).dx, at(s.first).dy);
      for (final p in s.skip(1)) {
        path.lineTo(at(p).dx, at(p).dy);
      }
      canvas.drawPath(path, ink);
    }
  }

  @override
  bool shouldRepaint(_StrokePainter old) => old.strokes != strokes;
}
